catholic_can_invite_scholar_trigger = {
	OR = {
		religion = catholic
		AND = {
			religion = orthodox
			BYZ = {
				has_estate_privilege = estate_church_union_of_the_churches
			}
		}
		AND = {
			OR = {
				religion = protestant
				religion = reformed
				religion = hussite
				religion = anglican
			}
			hre_religion_treaty = yes
			is_part_of_hre = yes
		}
	}
}
