assign_default_catholic_school_effect = {
	# Assign a starting school based on culture groups highlighted in 00_cultures.txt.
	if = {
		limit = {
			OR = {
				culture_group = french
				culture_group = iberian
			}
		}
		set_religious_school = {
			group = christian
			school = thomist_school
		}
	}
	else_if = {
		limit = {
			OR = {
				primary_culture = scottish
				culture_group = gaelic
			}
		}
		set_religious_school = {
			group = christian
			school = scotist_school
		}
	}
	else_if = {
		limit = {
			AND = {
				culture_group = british
				NOT = {
					primary_culture = scottish
				}
			}
		}
		set_religious_school = {
			group = christian
			school = nominalist_school
		}
	}
	else_if = {
		limit = {
			culture_group = latin
		}
		set_religious_school = {
			group = christian
			school = augustinian_school
		}
	}
	else_if = {
		limit = {
			OR = {
				current_age = age_of_reformation
				current_age = age_of_absolutism
				current_age = age_of_revolutions
			}
		}
		random_list = {
			1 = {
				set_religious_school = {
					group = christian
					school = molinist_school
				}
			}
			1 = {
				set_religious_school = {
					group = christian
					school = jansenist_school
				}
			}
		}
	}
	else = {
		set_religious_school = {
			group = christian
			school = thomist_school
		}
	}
}

assign_default_protestant_school_effect = {
	if = {
		limit = {
			culture_group = germanic
			NOT = {
				primary_culture = saxon
			}
		}
		set_religious_school = {
			group = christian
			school = wittenberg_school
		}
	}
	else_if = {
		limit = {
			culture_group = scandinavian
		}
		set_religious_school = {
			group = christian
			school = magdeburg_school
		}
	}
	else_if = {
		limit = {
			AND = {
				culture_group = british
				NOT = {
					primary_culture = scottish
				}
			}
			OR = {
				current_age = age_of_reformation
				current_age = age_of_absolutism
				current_age = age_of_revolutions
			}
		}
		set_religious_school = {
			group = christian
			school = philippist_school
		}
	}
	else_if = {
		limit = {
			primary_culture = saxon
			OR = {
				current_age = age_of_reformation
				current_age = age_of_absolutism
			}
		}
		set_religious_school = {
			group = christian
			school = gnesiolutheran_school
		}
	}
	else = {
		set_religious_school = {
			group = christian
			school = wittenberg_school
		}
	}
}

initialize_christian_schools_effect = {
	if = {
		limit = {
			NOT = {
				has_global_flag = christian_schools_initialized
			}
		}
		set_global_flag = christian_schools_initialized
	}
	every_country = {
		limit = {
			religion = catholic
			has_dlc = "Cradle of Civilization"
			OR = {
				NOT = {
					has_religious_school = yes
				}
				NOT = {
					is_catholic_school_country = yes
				}
			}
		}
		assign_default_catholic_school_effect = yes
	}
	every_country = {
		limit = {
			religion = protestant
			has_dlc = "Cradle of Civilization"
			OR = {
				NOT = {
					has_religious_school = yes
				}
				NOT = {
					is_lutheran_school_country = yes
				}
			}
		}
		assign_default_protestant_school_effect = yes
	}
	every_country = {
		limit = {
			religion = reformed
			has_dlc = "Cradle of Civilization"
			OR = {
				NOT = {
					has_religious_school = yes
				}
				NOT = {
					is_reformed_school_country = yes
				}
			}
		}
		assign_default_reformed_school_effect = yes
	}
	every_country = {
		limit = {
			religion = orthodox
			has_dlc = "Cradle of Civilization"
			OR = {
				NOT = {
					has_religious_school = yes
				}
				NOT = {

				}
			}
		}
		assign_default_orthodox_school_effect = yes
	}
	every_country = {
		limit = {
			religion = coptic
			has_dlc = "Cradle of Civilization"

				}
			}
		}
		assign_default_coptic_school_effect = yes
	}
