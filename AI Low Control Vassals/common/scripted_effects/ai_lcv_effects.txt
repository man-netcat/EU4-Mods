lcv_province_cede_to_best_neighbor = {
	save_event_target_as = lcv_province
	random_neighbor_province = {
		limit = {
			owner = {
				lcv_is_vassal_with_less_than_5_provinces = yes
			}
			owner = {
				event_target:lcv_province = {
					is_core = PREV
				}
			}
		}
		owner = {
			save_event_target_as = lcv_recipient
		}
		event_target:lcv_province = {
			cede_province = event_target:lcv_recipient
			lcv_change_government = yes
		}
	}
}

lcv_change_government = {
	event_target:lcv_recipient = {
		if = {
			limit = {
				overlord = {
					government = monarchy
				}
			}
			change_government = monarchy
		}
		else_if = {
			limit = {
				overlord = {
					government = republic
				}
			}
			change_government = republic
		}
		else_if = {
			limit = {
				overlord = {
					government = theocracy
				}
			}
			change_government = theocracy
		}
		else_if = {
			limit = {
				overlord = {
					government = tribal
				}
			}
			change_government = tribal
		}
	}
}
