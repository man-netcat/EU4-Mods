# Pirate Federation CBs

# CB to liberate island provinces and create pirate havens
cb_pirate_federation_liberate = {
	valid_for_subject = no
	
	prerequisites_self = {
		has_reform = pirate_republic_reform
		is_revolutionary = no
	}
	
	prerequisites = {
		# Target must own island provinces
		FROM = {
			any_owned_province = {
				province_is_on_an_island = yes
			}
			# Must be neighbor or nearby
			OR = {
				is_neighbor_of = ROOT
				any_owned_province = {
					any_neighbor_province = {
						owned_by = ROOT
					}
				}
				# Or within reasonable naval range
				ROOT = {
					any_owned_province = {
						has_port = yes
						any_neighbor_province = {
							owned_by = FROM
							has_port = yes
						}
					}
				}
			}
		}
	}
	
	war_goal = pirate_liberation_wargoal
	
	# Only allow releasing island provinces as pirate republics
	attacker_disabled_po = {
		po_demand_provinces
		po_revoke_cores
		po_become_vassal
		po_subjugate_vassal
		po_become_tributary_state
		po_subjugate_tributary_state
		po_form_personal_union
		po_transfer_vassals
	}
	
	ai_peace_desire = 10
}

# CB to invite island nations to join the pirate federation
cb_pirate_federation_invite = {
	valid_for_subject = no
	
	prerequisites_self = {
		has_reform = pirate_republic_reform
		is_revolutionary = no
	}
	
	prerequisites = {
		# Target must be entirely on islands
		FROM = {
			# All provinces must be on islands
			all_owned_province = {
				province_is_on_an_island = yes
			}
			# Not already a pirate republic
			NOT = { has_reform = pirate_republic_reform }
			# Not already in the federation
			NOT = { is_subject_of_type = pirate_federation_member }
			NOT = { overlord_of = ROOT }
			# Must be neighbor or nearby
			OR = {
				is_neighbor_of = ROOT
				any_owned_province = {
					any_neighbor_province = {
						owned_by = ROOT
					}
				}
				# Or within reasonable naval range
				ROOT = {
					any_owned_province = {
						has_port = yes
						any_neighbor_province = {
							owned_by = FROM
							has_port = yes
						}
					}
				}
			}
		}
	}
	
	war_goal = pirate_federation_wargoal
	
	# Only allow making them join the federation or taking some money
	attacker_disabled_po = {
		po_demand_provinces
		po_revoke_cores
		po_release_vassals
		po_release_annexed
		po_return_cores
		po_become_vassal
		po_subjugate_vassal
		po_become_tributary_state
		po_subjugate_tributary_state
		po_form_personal_union
		po_transfer_vassals
	}
	
	# Peace options allowed
	# Custom peace treaty will be defined separately
	
	ai_peace_desire = 10
}
