# Peace treaty to force a country to join the pirate federation
po_pirate_federation_join = {
	category = 6
	power_projection = vassalized_rival
	power_cost_base = 1
	prestige_base = 0.5
	ae_base = 0.5	# Lower AE than making a vassal
	warscore_cost = {
		all_provinces = 0.75
		no_provinces = 0
		owner_keeps = 0
		conquered = 0
		returned_core = 0
		released_nation = 0
		cancelled_subject = 0
		concede_colonial_region = 0
	}
	warscore_cap = 60
	requires_demand_independence = yes
	is_make_subject = yes
	requires_is_allowed = yes
	is_visible = {
		ROOT = {
			has_reform = pirate_republic_reform
		}
		FROM = {
			all_owned_province = {
				province_is_on_an_island = yes
			}
			NOT = {
				has_reform = pirate_republic_reform
			}
			NOT = {
				is_subject_of_type = pirate_federation_member
			}
		}
	}
	is_allowed = {
		ROOT = {
			has_reform = pirate_republic_reform
		}
		FROM = {
			all_owned_province = {
				province_is_on_an_island = yes
			}
		}
	}
	effect = {
		FROM = {
			# Convert to pirate republic if not already
			if = {
				limit = {
					NOT = {
						has_reform = pirate_republic_reform
					}
				}
				if = {
					limit = {
						NOT = {
							government = republic
						}
					}
					change_government = republic
				}
				add_government_reform = pirate_republic_reform
			}
		}
		# Create the subject relationship
		create_subject = {
			subject = FROM
			subject_type = pirate_federation_member
		}
	}
	ai_weight = {
		export_to_variable = {
			variable_name = ai_value
			value = 75
		}
	}
}

# Liberate island provinces as pirate republics that join the federation
po_release_pirate_vassals = {
	category = 6
	power_projection = vassalized_rival
	power_cost_base = 0.5
	prestige_base = 0.5
	ae_base = 0.4
	warscore_cost = {
		all_provinces = 0.75
		no_provinces = 0
		owner_keeps = 0
		conquered = 0
		returned_core = 0
		released_nation = 0
		cancelled_subject = 0
		concede_colonial_region = 0
	}
	warscore_cap = 60
	requires_demand_independence = yes
	is_make_subject = yes
	requires_is_allowed = yes
	is_visible = {
		always = yes
	}
	is_allowed = {
		custom_trigger_tooltip = {
			tooltip = PIRATE_LIBERATION_ALLOWED_TT
			ROOT = {
				has_reform = pirate_republic_reform
			}
		}
	}
	effect = {
		FROM = {
			set_country_flag = pirate_liberation_peace_defender
		}
		ROOT = {
			set_country_flag = pirate_liberation_peace_attacker
			save_event_target_as = pirate_liberator
		}
		FROM = {
			save_event_target_as = liberated_from
			country_event = {
				id = pirate_federation.10
				days = 1
			}
		}
	}
	ai_weight = {
		export_to_variable = {
			variable_name = ai_value
			value = 50
		}
	}
}
