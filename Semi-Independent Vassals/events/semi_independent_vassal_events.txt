namespace = semi_independent_vassal

# Vassal requests overlord to join defensive war
country_event = {
	id = semi_independent_vassal.1
	title = semi_independent_vassal.1.t
	desc = semi_independent_vassal.1.d
	picture = DIPLOMACY_eventPicture
	
	is_triggered_only = yes
	
	# Accept and join the war
	option = {
		name = semi_independent_vassal.1.a
		
		ai_chance = {
			factor = 50
			modifier = {
				factor = 2
				FROM = {
					reverse_has_opinion = {
						who = ROOT
						value = 150
					}
				}
			}
			modifier = {
				factor = 0.5
				is_bankrupt = yes
			}
			modifier = {
				factor = 0.5
				war_exhaustion = 5
			}
			modifier = {
				factor = 0.1
				war_exhaustion = 10
			}
		}
		
		join_all_defensive_wars_of = FROM
		
		# Improve relations
		FROM = {
			add_opinion = {
				who = ROOT
				modifier = overlord_joined_war
			}
		}
		
		# Cost favors if overlord can use them
		if = {
			limit = {
				has_dlc = "The Cossacks"
			}
			FROM = {
				add_trust = {
					who = ROOT
					value = 5
					mutual = yes
				}
			}
		}
	}
	
	# Decline
	option = {
		name = semi_independent_vassal.1.b
		
		ai_chance = {
			factor = 50
			modifier = {
				factor = 2
				is_bankrupt = yes
			}
			modifier = {
				factor = 1.5
				war_exhaustion = 5
			}
			modifier = {
				factor = 0.5
				FROM = {
					reverse_has_opinion = {
						who = ROOT
						value = 150
					}
				}
			}
		}
		
		FROM = {
			add_opinion = {
				who = ROOT
				modifier = overlord_refused_support
			}
			add_liberty_desire = 10
		}
	}
}

# Overlord chooses to enforce peace for vassal
country_event = {

	title = semi_independent_vassal.2.t
	desc = semi_independent_vassal.2.d
	picture = DIPLOMACY_eventPicture
	
	is_triggered_only = yes
	
	# Join the defensive war
	option = {
		name = semi_independent_vassal.2.a
		
		join_all_defensive_wars_of = event_target:defended_vassal
		
		event_target:defended_vassal = {
			add_opinion = {
				who = ROOT
				modifier = overlord_joined_war
			}
		}
	}
}
