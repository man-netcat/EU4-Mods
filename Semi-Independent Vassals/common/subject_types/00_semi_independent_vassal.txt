# Semi-Independent Vassal Subject Type
# A vassal type that maintains significant autonomy and can wage its own wars

semi_independent_vassal = {
	copy_from = vassal
	
	# Graphics:
	sprite = GFX_icon_vassal
	diplomacy_overlord_sprite = GFX_diplomacy_leadvassal
	diplomacy_subject_sprite = GFX_diplomacy_amvassal

	# Properties:
	# Key difference: They do NOT automatically join overlord's wars
	joins_overlords_wars = no
	
	# Can be co-belligerented by enemies (defensive wars)
	can_be_co_belligerented = yes
	
	# Overlord can call them to wars using favors
	must_accept_cta_from_overlord = yes
	favors_cost_to_join_offensive_wars = 10
	favors_cost_to_join_defensive_wars = 5
	
	# Cannot be annexed - they're too independent for that
	can_be_annexed = no
	
	# Trade and economic properties
	transfer_trade_if_merchant_republic = yes
	pays_overlord = 0.5  # Pays less than regular vassals due to semi-independence
	forcelimit_to_overlord = 0.05  # Provides less military support
	
	# Power projection
	has_power_projection = yes
	
	# They can fight, ally, rival, and marry freely - fully independent foreign policy
	can_fight = {
		independent_nations = yes
		other_overlord = all
	}
	can_rival = {
		independent_nations = yes
		other_overlord = all
	}
	can_ally = {
		independent_nations = yes
		same_overlord = semi_independent_vassal
		other_overlord = all
	}
	can_marry = {
		independent_nations = yes
		same_overlord = semi_independent_vassal
		other_overlord = all
	}
	
	# Overlord protections
	overlord_protects_external = no  # Overlord is not automatically called
	overlord_enforce_peace_attacking = yes  # Overlord CAN enforce peace when vassal is attacked
	
	# Can use their own claims
	can_use_claims = yes
	overlord_can_fabricate_for = yes
	
	# Rebels and help
	gets_help_with_rebels = yes
	share_rebel_popup = yes
	
	# Liberty desire - LOWER base due to greater autonomy granted
	# They're content with their independence, so much lower base liberty desire
	base_liberty_desire = -25.0
	liberty_desire_development_ratio = 0.15
	liberty_desire_same_dynasty = -10.0
	
	# Military focus - they maintain their own military
	military_focus = 1.3
	
	# Can have trade companies
	forms_trade_companies = no
	
	# Can concentrate development
	can_concentrate_development = yes
	
	# Max government rank
	max_government_rank = 2  # Can be kingdoms
	
	# Diplomatic slot
	takes_diplo_slot = yes
	
	# Trust
	trust_on_start = 40
	
	# Relative power class
	# They CAN cooperate for independence, but the low base liberty desire means
	# they're much less likely to actually want to
	relative_power_class = 1
	should_quit_wars_on_activation = no  # Don't quit wars when becoming semi-independent
	
	# Subject Interactions:
	# (continuous)
	embargo_rivals = yes
	support_loyalists = yes
	subsidize_armies = yes
	scutage = no  # They don't pay scutage since they fight their own wars
	send_officers = yes
	divert_trade = yes
	
	# (immediate)
	placate_rulers = yes
	place_relative_on_throne = yes
	enforce_religion = yes
	customize_subject = no
	replace_governor = no
	grant_province = yes
	enforce_culture = yes
	siphon_income = no
	fortify_march = no
	seize_territory = yes
	start_colonial_war = no
	grant_core_claim = yes
	sacrifice_ruler = no
	sacrifice_heir = yes
	takeondebt = yes
	bestow_gifts = yes
	send_additional_troops = no
	demand_artifacts = no
	demand_additional_tribute = no
	force_seppuku = no
	press_sailors = no
	contribute_to_capital = no
	force_isolation = no
	return_land = yes
	conscript_general = no
	knowledge_sharing = yes
	block_settlement_growth = yes
	allow_settlement_growth = yes
	change_colonial_type = no
	upgrade_subject_type = no
	seize_court_resources = no
	request_extra_levies = no
	grant_administrative_autonomy = no
	
	# Modifiers:
	modifier_subject = clear
	modifier_subject = {
		modifier = semi_independent_vassal_nation
	}
	modifier_overlord = clear
	modifier_overlord = {
		modifier = semi_independent_vassal_overlord
	}
	
	overlord_opinion_modifier = null
	subject_opinion_modifier = null
}
