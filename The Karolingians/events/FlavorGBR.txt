# Religious Stance of the current King
country_event = {
	id = flavor_gbr.102
	title = flavor_gbr.102.t
	desc = flavor_gbr.102.desc
	picture = HENRY_eventPicture
	is_triggered_only = yes	#Fires whenever you have a new monarch; religious policy of the king
	trigger = {
		has_country_modifier = eng_the_age_of_reformation
	}
	mean_time_to_happen = {
		days = 1
	}
	goto = event_province
	immediate = {
		hidden_effect = {
			random_owned_province = {
				limit = {
					religion = ROOT
					is_in_capital_area = yes
					NOT = { has_province_modifier = religious_center }
				}
				save_event_target_as = event_province
			}
		}
	}
	#We must follow the reformation
	option = {
		name = flavor_gbr.102.a
		add_church_power = 50
		add_fervor = 25
		if = {
			limit = {
				religion = catholic
			}
			add_papal_influence = -200
			add_estate_loyalty = {
				estate = estate_church
				loyalty = -15
			}
		}
		if = {
			limit = {
				religion = catholic
			}
			add_ruler_modifier = {
				name = eng_reformer_ruler_for_catholic_gb
				duration = -1
				desc = until_end_of_age_of_reformation_desc
			}
		}
		else = {
			add_ruler_modifier = {
				name = eng_reformer_ruler
				duration = -1
				desc = until_end_of_age_of_reformation_desc
			}
		}
		if = {
			limit = {
				NOT = { religion = catholic }
			}
			event_target:event_province = {
				change_religion = catholic
			}
		}
		else = {
			if = {
				limit = { is_religion_enabled = anglican }
				event_target:event_province = {
					change_religion = anglican
				}
			}
			else = {
				event_target:event_province = {
					change_religion = protestant
				}
			}
		}
		custom_tooltip = flavor_gbr.102.a.at
		if = {
			limit = {
				NOT = { ruler_religion = protestant }
				NOT = { ruler_religion = reformed }
				NOT = { ruler_religion = anglican }
				NOT = { ruler_religion = hussite }
			}
			custom_tooltip = flavor_gbr.102.a.tt
			hidden_effect = {
				every_known_country = {
					limit = {
						religion = catholic
					}
					add_opinion = {
						who = ROOT
						modifier = eng_turned_back_to_the_pope
					}
				}
			}
			set_ruler_religion = ROOT
		}
		hidden_effect = {
			every_known_country = {
				limit = {
					OR = {
						religion = protestant
						religion = reformed
						religion = anglican
						religion = hussite
					}
				}
				add_opinion = {
					who = ROOT
					modifier = eng_pro_reformation
				}
				reverse_add_opinion = {
					who = ROOT
					modifier = eng_pro_reformation
				}
			}
		}
		ai_chance = {
			factor = 1
			modifier = {
				factor = 0
				NOT = { ruler_religion = protestant }
				NOT = { ruler_religion = reformed }
				NOT = { ruler_religion = anglican }
				NOT = { ruler_religion = hussite }
			}
			modifier = {
				factor = 0
				ruler_has_personality = tolerant_personality
			}
		}
	}
	#We must return to Catholicism
	option = {
		name = flavor_gbr.102.b
		add_papal_influence = 50
		if = {
			limit = {
				OR = {
					religion = protestant
					religion = reformed
					religion = anglican
					religion = hussite
				}
			}
