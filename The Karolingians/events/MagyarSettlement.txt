# AI auto-formation (CK3's game-rule events govern the AI migration while
# the player takes the decision by hand): fires kar_form_hungary's effect
# for AI MGY the moment its allow conditions hold. The land/capital/rank
# numbers mirror decisions/FormHUN.txt (generated from the formables row);
# the shared body lives in common/scripted_effects/MagyarEffects.txt.
# The settlement event above then fires for the new HUN in turn.
country_event = {
	id = magyar_settlement.2
	title = magyar_settlement_2_title
	desc = magyar_settlement_2_desc
	picture = HORDE_ON_HORSEBACK_eventPicture

	trigger = {
		ai = yes
		tag = MGY
		NOT = { exists = HUN }
		NOT = { has_country_flag = kar_form_hungary }
		government_rank = 3
		is_at_war = no
		is_free_or_tributary_trigger = yes
		owns = 153
		num_of_owned_provinces_with = {
			value = 12
			kar_form_hungary_provinces_trigger = yes
		}
	}

	mean_time_to_happen = {
		months = 1
	}

	option = {
		name = magyar_settlement_2_a
		ai_chance = { factor = 100 }
		kar_form_hungary_crown_effect = yes
	}
}

# Migration-war victory (CK3's hungarian_migration_cb on_victory: the whole
# basin plus the crown at once). Fired from on_war_won for an MGY victory
# over a basin holder; the peace deal's own spoils land first, then this
# completes the de-jure basin. Simplification against CK3, which seizes only
# the defender's share: the event takes every unheld basin province.
country_event = {
	id = magyar_settlement.3
	title = magyar_settlement_3_title
	desc = magyar_settlement_3_desc
	picture = HORDE_ON_HORSEBACK_eventPicture

	is_triggered_only = yes

	trigger = {
		tag = MGY
		NOT = { exists = HUN }
		NOT = { has_country_flag = kar_form_hungary }
	}

	option = {
		name = magyar_settlement_3_a
		ai_chance = { factor = 100 }
		alfold_area = {
			limit = { NOT = { owned_by = ROOT } }
			add_core = ROOT
			cede_province = ROOT
		}
		southern_transylvania_area = {
			limit = { NOT = { owned_by = ROOT } }
			add_core = ROOT
			cede_province = ROOT
		}
		transdanubia_area = {
			limit = { NOT = { owned_by = ROOT } }
			add_core = ROOT
			cede_province = ROOT
		}
		transylvania_area = {
			limit = { NOT = { owned_by = ROOT } }
			add_core = ROOT
			cede_province = ROOT
		}
		kar_form_hungary_crown_effect = yes
	}

	option = {
		name = magyar_settlement_3_b
		ai_chance = { factor = 0 }
		add_prestige = 10
	}
}
