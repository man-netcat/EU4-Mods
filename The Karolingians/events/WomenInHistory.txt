# Gráinne Ní Mháille
country_event = {
	id = women_history_events.22
	title = women_history_events.22.t
	desc = women_history_events.22.d
	picture = wihgfx_FEMALES_eventPicture

	trigger = {
		OR = {
			AND = {
				has_government_attribute = is_pirate_republic_reform
				has_global_flag = grace_omalley_flag
			}
			AND = {
				owns = 375
				owns = 376
				is_year = 1530
				NOT = {
					is_year = 1605
				}
			}
		}
		NOT = {
			has_global_flag = wih_grainnenimhaille_flag
		}
	}
	
	mean_time_to_happen = {
		months = 200
	}
	
	immediate = {
		hidden_effect = {
			set_global_flag = wih_grainnenimhaille_flag
		}
	}
	
	option = {
		name = women_history_events.22.a
		trigger = {
			NOT = { has_global_flag  = grace_omalley_flag }
		}
		add_meritocracy_effect = yes
		define_advisor = {
			female = yes
			type = commandant
			culture = irish
			skill = 2
			name = "Grainne Ni Mhaille"
			discount = yes
		}
	}
	option = {
		name = women_history_events.22.b
		trigger = {
			NOT = { has_global_flag  = grace_omalley_flag }
		}
		define_admiral = {
			name = "Grainne Ni Mhaille"
			shock = 4
			fire = 4
			manuever = 3
			female = yes
		}
	}
	option = {
		name = women_history_events.22.c
		trigger = {
			NOT = { has_global_flag  = grace_omalley_flag }
		}
		add_prestige = 10
	}
	option = {
		name = women_history_events.22.da
		trigger = {
			has_global_flag  = grace_omalley_flag
		}
		define_ruler = {
			name = "Grainne Ni Mhaille"
			adm = 3
			dip = 4
			mil = 5
			culture = irish
		}
		if = {
			limit = {
				has_spare_flagship_capacity = yes
				capital_scope = {
					has_port = yes
				}
			}
			capital_scope = {
				create_flagship = {	
					name = "White Seahorse"
					type = heavy_ship
				}
			}
		}
		else_if = {
			limit = {
				capital_scope = {
					has_port = yes
				}
			}
			capital_scope = {
				create_named_ship = {
					name = "White Seahorse"
					type = heavy_ship
				}
			}
		}
	}
}
