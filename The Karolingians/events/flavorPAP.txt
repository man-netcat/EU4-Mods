# The Jesuits return

country_event = {
	id = flavor_pap.14
	title = "flavor_pap.EVTNAME14"
	desc = "flavor_pap.EVTDESC14"
	picture = RELIGION_eventPicture
	
	is_triggered_only = yes
	
	immediate = {
		hidden_effect = {
			random_owned_province = {
				limit = {
					is_capital = yes
				}
				save_event_target_as = current_china_capital
			}
		}
	}
	
	option = {
		name = "flavor_pap.EVTOPTA14" # Yes, we may benefit from their learning
		random_owned_province = {
			limit = {
				has_port = yes
				is_reformation_center = no
			}
			change_religion = catholic
			add_permanent_province_modifier = {
				name = religious_zeal_at_conv
				duration = 7300
			}
			if = {
				limit = { NOT = { renaissance = 100 } }
				add_institution_embracement = {
					which = renaissance
					value = 50
				}
			}
		}
		add_dip_power = 100
		add_mil_power = 100
		add_adm_power = 100
		every_known_country = {
			limit = {
				religion = catholic
				has_discovered = event_target:current_china_capital
			}
			add_opinion = {
				who = ROOT
				modifier = opinion_accepted_jesuits
			}
		}
	}
	option = {
		name = "flavor_pap.EVTOPTB14"
		add_stability_or_adm_power = yes
	}
}

# Kirishitan return!
country_event = {
	id = flavor_pap.15
	title = "flavor_pap.EVTNAME15"
	desc = "flavor_pap.EVTDESC15"
	picture = RELIGION_eventPicture
	
	is_triggered_only = yes
	
	immediate = {
		if = {
			limit = {
				NOT = { num_of_cities = 8 }
				all_owned_province = {
					NOT = { religion = catholic }
				}
			}
			random_owned_province = {
				limit = {
					culture_group = japanese_g
				}
				change_religion = catholic
			}
		}
		
		else_if = {
			limit = {
				num_of_cities = 8
				NOT = { num_of_cities = 15 }
				all_owned_province = {
					NOT = { religion = catholic }
				}
			}
			random_owned_province = {
				limit = {
					culture_group = japanese_g
					NOT = { religion = catholic }
				}
				change_religion = catholic
			}
			random_owned_province = {
				limit = {
					culture_group = japanese_g
					NOT = { religion = catholic }
				}
				change_religion = catholic
			}
		}
		
		else_if = {
			limit = {
				num_of_cities = 15
				all_owned_province = {
					NOT = { religion = catholic }
				}
			}
			random_owned_province = {
				limit = {
					culture_group = japanese_g
					NOT = { religion = catholic }
				}
				change_religion = catholic
			}
			random_owned_province = {
				limit = {
					culture_group = japanese_g
					NOT = { religion = catholic }
				}
				change_religion = catholic
			}
			random_owned_province = {
				limit = {
					culture_group = japanese_g
					NOT = { religion = catholic }
				}
				change_religion = catholic
			}
		}
			
		hidden_effect = {
			random_owned_province = {
				limit = {
					culture_group = japanese_g
					religion = catholic
					can_spawn_rebels = catholic_rebels
				}
				set_province_flag = jap_kirishitan_flag
			}
			if = {
				limit = {
					num_of_cities = 8
				}
				random_owned_province = {
					limit = {
						culture_group = japanese_g
						religion = catholic
						can_spawn_rebels = catholic_rebels
						NOT = { has_province_flag = jap_kirishitan_flag }
					}
					set_province_flag = jap_kirishitan_flag
				}
			}
			if = {
				limit = {
					num_of_cities = 15
				}
				random_owned_province = {
					limit = {
						culture_group = japanese_g
						religion = catholic
						can_spawn_rebels = catholic_rebels
						NOT = { has_province_flag = jap_kirishitan_flag }
					}
					set_province_flag = jap_kirishitan_flag
				}
			}
		}
	}
	
	option = {
		name = "flavor_pap.EVTOPTA15" # Stop the missionaries!
		every_owned_province = {
			limit = {
				has_province_flag = jap_kirishitan_flag
			}
			spawn_rebels = { 
				type = catholic_rebels 
				size = 2 
			}
			clr_province_flag = jap_kirishitan_flag
		}
	}
	option = {
		name = "flavor_pap.EVTOPTB15" # Let them spread their faith
		add_prestige = -25
		add_stability = -1
		every_owned_province = {
			limit = {
				has_province_flag = jap_kirishitan_flag
			}
			add_province_modifier = {
				name = add_unrest_5_modifier
				duration = 1825
			}
			clr_province_flag = jap_kirishitan_flag
		}
	}
}
