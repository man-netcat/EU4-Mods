#End of the Black Army
country_event = {
	id = flavor_hun.18
	title = flavor_hun.18.t
	desc = flavor_hun.18.d
	picture = MILITARY_CAMP_eventPicture

	fire_only_once = yes

	trigger = {
		current_age = age_of_reformation
		has_country_modifier = hun_black_army
		NOT = { has_reform = black_army_reform }
	}

	mean_time_to_happen = {
		months = 12
	}

	option = {
		name = flavor_hun.18.a
		remove_country_modifier = hun_black_army
		add_army_tradition = 20
		add_absolutism = 10
		add_mil_power = 75
		if = {
			limit = {
				has_estate = estate_nobles
			}
			custom_tooltip = end_of_black_army_tooltip
		}
	}
	option = { # new option, keep the black army
		name = flavor_hun.18.b
		trigger = {
			is_subject = no
			OR = {
				has_dlc = "Emperor"
				has_dlc = "Domination"
			}
		}
		set_country_flag = black_army_reform_flag
		add_government_reform = black_army_reform
		if = {
			limit = {
				has_estate = estate_nobles
			}
			add_estate_loyalty = {
				estate = estate_nobles
				loyalty = -15
			}
		}
	}
}

#Mutiny in the Black Army
country_event = {
	id = flavor_hun.19
	title = flavor_hun.19.t
	desc = flavor_hun.19.d
	picture = MILITARY_CAMP_eventPicture

	trigger = {
		has_country_modifier = hun_black_army
		OR = {
			NOT = { has_country_flag = had_black_army_mutiny }
			had_country_flag = { flag = had_black_army_mutiny days = 7300 }
		}
		is_at_war = yes
		num_of_cities = 5
		any_owned_province = {
			is_capital = no
			is_in_capital_area = yes
			NOT = { has_province_modifier = mercenary_concessions }
		}
		num_of_mercenaries = 1
		NOT = { has_reform = black_army_reform }
	}

	immediate = {
		hidden_effect = {
			set_country_flag = had_black_army_mutiny
			random_owned_province = {
				limit = {
					is_capital = no
					is_in_capital_area = yes
					NOT = { has_province_modifier = mercenary_concessions }
				}
				save_event_target_as = mercenary_mutiny_province
			}
			random_owned_province = {
				limit = {
					is_capital = no
					is_in_capital_area = yes
					development = 6
					NOT = { has_province_modifier = mercenary_concessions }
				}
				save_event_target_as = mercenary_mutiny_province
			}
			random_owned_province = {
				limit = {
					is_capital = no
					is_in_capital_area = yes
					development = 10
					NOT = { has_province_modifier = mercenary_concessions }
				}
				save_event_target_as = mercenary_mutiny_province
			}
		}
	}

	mean_time_to_happen = {
		months = 80
	}

	option = {
		name = flavor_hun.19.a
		event_target:mercenary_mutiny_province = {
			spawn_rebels = {
				type = noble_rebels
				size = 2
			}
			add_devastation = 35
		}
	}
	option = {
		name = flavor_hun.19.b
		event_target:mercenary_mutiny_province = {
			add_province_modifier = {
				name = "mercenary_concessions"
				duration = 7300
			}
		}
	}
}
