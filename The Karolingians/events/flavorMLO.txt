namespace = new_flavour_mlo


# Silk in Como?
province_event = {
	id = new_flavour_mlo.15
	title = new_flavour_mlo.15.name
	desc = new_flavour_mlo.15.desc
	picture = TRADE_GOODS_PLANTATION_GOODS_IN_WAREHOUSE_eventPicture
	
	is_triggered_only = yes
	
	goto = 4740

	immediate = {
		hidden_effect = {
			owner = {
				set_saved_name = {
					key = mlo_silk_guy
					type = advisor
					scope = this
				}
			}
		}
	}
	
	option = {
		name = new_flavour_mlo.15.a
		goto = 4740
		owner = {
			add_dip_power = -100
		}

		add_base_production = 2
		change_trade_goods = silk

		ai_chance = {
			factor = 1
		}
	}
	option = {
		name = new_flavour_mlo.15.b #Channel his inventive energies elsewhere.

		owner = {
			if = {
				limit = {
					NOT = { monthly_income = 15 }
				}
				define_advisor = {
					type = natural_scientist
					name = mlo_silk_guy
					skill = 1
					discount = yes
				}
			}
			else_if = {
				limit = {
					NOT = { monthly_income = 25 }
				}
				define_advisor = {
					type = natural_scientist
					name = mlo_silk_guy
					skill = 2
					discount = yes
				}
			}
			else = {
				define_advisor = {
					type = natural_scientist
					name = mlo_silk_guy
					skill = 3
					discount = yes
				}
			}
		}
		
		ai_chance = {
			factor = 0
		}
	}
	after = {
		owner = {
			clear_saved_name = mlo_silk_guy
		}
	}
}
