# Additive file: merged with vanilla's common/on_actions/00_on_actions.txt
# (same-key on_action blocks coexist across files, like decisions), so this
# carries only our guard. If the engine ever stops merging, this file must
# become a full vanilla copy instead.
# ROOT = winning country, FROM = loser country.
on_war_won = {
	if = {
		limit = {
			ROOT = {
				tag = MGY
				NOT = { exists = HUN }
				NOT = { has_country_flag = kar_form_hungary }
			}
			FROM = {
				OR = {
					tag = BUL
					tag = TRA
					tag = VSG
				}
			}
		}
		ROOT = { country_event = { id = magyar_settlement.3 } }
	}
}

# Migration-war chest on game start exactly (no MTTH, no event): the clans
# melt down their treasures before the first muster. ROOT is each country
# at startup (vanilla's own per-country setup pattern).
on_startup = {
	if = {
		limit = { tag = MGY }
		add_treasury = 500
		add_army_tradition = 20
		create_unit_forcelimit_percentage = {
			type = cavalry
			special_unit_category = magyar_horse
			force_limit_percentage = 1.0
		}
	}
}
