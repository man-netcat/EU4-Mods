#Livonian strong duchies: LIV-only override of the vanilla noble privilege.
#Mirrors estate_nobles_french_strong_duchies: LIV can take this in place of
#the vanilla Strong Duchies, and it reduces liberty desire for its bishopric
#vassals (theocracies) by 25% while allowing one extra vassal.

estate_nobles_livonian_strong_duchies = {
	icon = privilege_royal_authority
	max_absolutism = -10
	conditional_modifier = { trigger = { has_absolutism_reduction_for_estate_privileges = yes } modifier = { max_absolutism = 2 } }
	loyalty = 0.1
	influence = 0.1
	check_valid_when_tag_switching = no
	is_valid = {
		OR = {
			has_estate_privilege = estate_nobles_livonian_strong_duchies
			AND = {
				tag = LIV
				NOT = { has_estate_privilege = estate_nobles_strong_duchies }
			}
		}
		any_subject_country = {
			OR = {
				is_subject_of_type = vassal
				is_subject_of_type = march
				is_subject_of_type = client_march
				is_subject_of_type = client_vassal
				is_subject_of_type = personal_union
				is_subject_of_type = incorporated_vassal
				is_subject_of_type = integrated_personal_union
			}
			government = theocracy
		}
	}
	can_select = {
		custom_trigger_tooltip = {
			tooltip = estate_nobles_strong_duchies_trigger_tooltip
			calc_true_if = {
				amount = 2
				all_subject_country = {
					OR = {
						is_subject_of_type = vassal
						is_subject_of_type = march
						is_subject_of_type = client_march
						is_subject_of_type = client_vassal
						is_subject_of_type = personal_union
						is_subject_of_type = incorporated_vassal
						is_subject_of_type = integrated_personal_union
					}
					government = theocracy
				}
			}
		}
		NOT = { has_estate_privilege = estate_eunuchs_delegate_subject_diplomacy_privilege }
	}
	penalties = {
	}
	benefits = {
		diplomatic_upkeep = 3
		reduced_liberty_desire_on_same_continent = 25
	}
	ai_will_do = {
		factor = 2
		modifier = {
			factor = 3
			calc_true_if = {
				amount = 1
				all_subject_country = {
					is_vassal = yes
					government = theocracy
				}
			}
		}
		modifier = {
			factor = 2
			calc_true_if = {
				amount = 2
				all_subject_country = {
					is_vassal = yes
					government = theocracy
				}
			}
		}
	}
}
