# Historical European Tags - custom government names
# The 000_ prefix makes this file load before vanilla 00_/01_/zz_default
# government_names blocks. The base game picks the first valid block it
# finds, so our tag-scoped blocks win over default_monarchy etc.

# The Principality of Orange. A sovereign state within the Holy Roman
# Empire, ruled by the princes of the House of Orange. Not in the HRE
# de facto, so vanilla default_monarchy would give it "Duchy"; we give
# it its historical title.

orange_principality = {
	rank = {
		1 = PRINCIPALITY
		2 = KINGDOM
		3 = EMPIRE
	}

	ruler_male = {
		1 = PRINCE
		2 = KING
		3 = EMPEROR
	}

	ruler_female = {
		1 = PRINCESS
		2 = QUEEN
		3 = EMPRESS
	}

	consort_male = {
		1 = PRINCE_CONSORT
		2 = KING_CONSORT
		3 = EMPEROR_CONSORT
	}

	consort_female = {
		1 = CONSORT
		2 = QUEEN_CONSORT
		3 = EMPRESS_CONSORT
	}

	heir_male = {
		1 = HEIR
		2 = HEIR
		3 = HEIR
	}

	heir_female = {
		1 = HEIR_fem
		2 = HEIR_fem
		3 = HEIR_fem
	}

	trigger = {
		tag = ORA
		government = monarchy
	}
}
