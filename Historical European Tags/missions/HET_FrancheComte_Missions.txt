# Franche-Comte: mission tree toward the Kingdom of Arles.
#
# The tree is deliberately one mission per area that decisions/FormArles.txt
# requires, in the order that keeps the war fronts survivable. FormArles
# demands all 16 provinces of provence, bourgogne, savoy_dauphine and
# romandie owned AND cored, so each mission here completes one of those
# areas exactly, and hands out the claims that make the next one reachable.
#
# FRC starts as a two-province county (193 Franche-Comte, 4764 Salins) inside
# Burgundy, so the opening mission is about breaking out, not about conquest.
# Note FRC holds cores on 193 and 4764 but both are Burgundian in 1444 and
# Burgundy holds the third (192 Dijon), hence taking Dijon is the tree's spine.

frc_missions_to_arles = {
	slot = 1
	generic = no
	ai = yes
	# No DLC gate: decisions/FormArles.txt is DLC-free, so gating the tree
	# would hide it from vanilla installs that have no DLC at all.
	potential = {
		tag = FRC
		NOT = { map_setup = map_setup_random }
	}
	has_country_shield = yes

	# --- Opening: survive as a county, then take the Burgundian hinge ----

	frc_secure_the_jura = {
		icon = claim_the_borderlands
		position = 1
		required_missions = {  }

		provinces_to_highlight = {
			OR = {
				province_id = 193	# Franche-Comte
				province_id = 4764	# Salins
			}
			NOT = { owned_by = root }
		}

		trigger = {
			193 = {
				owned_by = ROOT
				is_core = ROOT
			}
			4764 = {
				owned_by = ROOT
				is_core = ROOT
			}
		}

		effect = {
			add_prestige = 10
			add_stability = 0.05
			# Dijon is the hinge of bourgogne_area and the seat of Burgundy.
			# Without a claim on it the Arles form is unreachable.
			192 = {
				add_permanent_claim = ROOT
			}
		}
		ai_weight = {
			factor = 1	#Already here
		}
	}

	frc_take_dijon = {
		icon = conquer_burgundy
		position = 2
		required_missions = { frc_secure_the_jura }

		provinces_to_highlight = {
			area = bourgogne_area
			NOT = { owned_by = root }
			NOT = { is_core = root }
		}

		trigger = {
			bourgogne_area = {
				type = all
				owned_by = ROOT
				is_core = ROOT
			}
		}

		effect = {
			add_prestige = 15
			add_dip_power = 50
			# The next two fronts. Provence answers the Dauphine and the
			# Romandie lies between Savoy and Burgundy.
			provence_area = {
				limit = {
					NOT = { is_core = ROOT }
					NOT = { is_permanent_claim = ROOT }
				}
				add_permanent_claim = ROOT
			}
			savoy_dauphine_area = {
				limit = {
					NOT = { is_core = ROOT }
					NOT = { is_permanent_claim = ROOT }
				}
				add_permanent_claim = ROOT
			}
		}
		ai_weight = {
			factor = 2	#Natural conquest
		}
	}

	# --- Parallel fronts: the Dauphine and the sea road -------------------

	frc_conquer_dauphine = {
		icon = conquer_east_francia
		position = 3
		required_missions = { frc_take_dijon }

		provinces_to_highlight = {
			area = savoy_dauphine_area
			NOT = { owned_by = root }
			NOT = { is_core = root }
		}

		trigger = {
			savoy_dauphine_area = {
				type = all
				owned_by = ROOT
				is_core = ROOT
			}
		}

		effect = {
			add_prestige = 20
			add_manpower = 5000
			# Savoy sits in two of the four areas, so it is claimed from both
			# directions before the Romandie push.
			savoy_dauphine_area = {
				limit = {
					NOT = { is_core = ROOT }
					NOT = { is_permanent_claim = ROOT }
				}
				add_permanent_claim = ROOT
			}
			romandie_area = {
				limit = {
					NOT = { is_core = ROOT }
					NOT = { is_permanent_claim = ROOT }
				}
				add_permanent_claim = ROOT
			}
		}
		ai_weight = {
			factor = 3	#Required for Arles
		}
	}

	frc_take_provence = {
		icon = mission_italian_condottiere
		position = 3
		required_missions = { frc_take_dijon }

		provinces_to_highlight = {
			area = provence_area
			NOT = { owned_by = root }
			NOT = { is_core = root }
		}

		trigger = {
			provence_area = {
				type = all
				owned_by = ROOT
				is_core = ROOT
			}
		}

		effect = {
			add_prestige = 20
			add_manpower = 5000
			# Avignon (202) is the FormArles capital target and the last
			# province the papacy holds in the set.
			202 = {
				limit = {
					NOT = { is_core = ROOT }
					NOT = { is_permanent_claim = ROOT }
				}
				add_permanent_claim = ROOT
			}
		}
		ai_weight = {
			factor = 3	#Required for Arles
		}
	}

	# --- The Romandie: the fourth and hardest area -----------------------

	frc_take_romandie = {
		icon = mission_landsknecht_soldier
		position = 5
		required_missions = { frc_conquer_dauphine }

		provinces_to_highlight = {
			area = romandie_area
			NOT = { owned_by = root }
			NOT = { is_core = root }
		}

		trigger = {
			romandie_area = {
				type = all
				owned_by = ROOT
				is_core = ROOT
			}
		}

		effect = {
			add_prestige = 25
			add_mil_power = 100
			add_stability = 0.05
			# Geneva (4720) is the last non-Savoy, non-Swiss holdout in the
			# area and the one that can swallow the war effort if left alone.
			4720 = {
				limit = {
					NOT = { is_core = ROOT }
					NOT = { is_permanent_claim = ROOT }
				}
				add_permanent_claim = ROOT
			}
		}
		ai_weight = {
			factor = 4	#Required for Arles
		}
	}

	# --- Capstone: all four areas held, the kingdom is within reach ------

	frc_crown_of_arles = {
		icon = carolingian_empire
		position = 7
		required_missions = {
			frc_take_dijon
			frc_conquer_dauphine
			frc_take_provence
			frc_take_romandie
		}

		provinces_to_highlight = {
			OR = {
				area = provence_area
				area = bourgogne_area
				area = savoy_dauphine_area
				area = romandie_area
			}
			NOT = { owned_by = root }
			NOT = { is_core = root }
		}

		# Mirrors the FormArles allow block exactly: all sixteen provinces,
		# owned and cored, and not already at war. The kingdom is the reward
		# for holding the Arelat whole, not for any single battlefield.
		trigger = {
			is_at_war = no
			provence_area = {
				type = all
				owned_by = ROOT
				is_core = ROOT
			}
			bourgogne_area = {
				type = all
				owned_by = ROOT
				is_core = ROOT
			}
			savoy_dauphine_area = {
				type = all
				owned_by = ROOT
				is_core = ROOT
			}
			romandie_area = {
				type = all
				owned_by = ROOT
				is_core = ROOT
			}
		}

		effect = {
			add_prestige = 25
			set_government_rank = 2
			add_country_modifier = {
				name = "centralization_modifier"
				duration = 7300
			}
			# The Italian lands the kingdom held before the crown passed to
			# the HRE. FormArles grants these too; the tree grants them early
			# so the player can see the shape of the claim before forming.
			piedmont_area = {
				limit = {
					NOT = { is_core = ROOT }
					NOT = { is_permanent_claim = ROOT }
				}
				add_permanent_claim = ROOT
			}
			lombardy_area = {
				limit = {
					NOT = { is_core = ROOT }
					NOT = { is_permanent_claim = ROOT }
				}
				add_permanent_claim = ROOT
			}
			po_valley_area = {
				limit = {
					NOT = { is_core = ROOT }
					NOT = { is_permanent_claim = ROOT }
				}
				add_permanent_claim = ROOT
			}
		}
		ai_weight = {
			factor = 5	#Natural conquest
		}
	}
}
