# 1 - The Strange Sickness (post-contact epidemic reaching native lands)
country_event = {
	id = zzz_sotnw.1
	title = "zzz_sotnw.1.t"
	desc = "zzz_sotnw.1.d"
	picture = PLAGUE_eventPicture

	trigger = {
		is_random_new_world = no
		NOT = { has_country_flag = zzz_sotnw_sickness }
		OR = {
			capital_scope = { continent = north_america }
			capital_scope = { continent = south_america }
		}
		is_year = 1450
	}

	mean_time_to_happen = {
		months = 300
		modifier = {
			factor = 2
			any_neighbor_country = { technology_group = western }
		}
	}

	option = {
		name = "zzz_sotnw.1.a"
		ai_chance = { factor = 70 }
		set_country_flag = zzz_sotnw_sickness
		add_war_exhaustion = 1
		add_manpower = -5000
	}

	option = {
		name = "zzz_sotnw.1.b"
		ai_chance = { factor = 30 }
		set_country_flag = zzz_sotnw_sickness
		add_stability = -1
		add_prestige = 10
	}
}
