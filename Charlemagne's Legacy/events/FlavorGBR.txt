	desc = flavor_gbr.113.desc
	picture = ANGRY_MOB_eventPicture
	is_triggered_only = yes
	goto = revolt_province
	immediate = {
		hidden_effect = {
			random_owned_province = {
				limit = {
					is_overseas = no
					is_in_capital_area = yes
				}
				save_event_target_as = revolt_province
			}
		}
	}
	
	#Let us turn to aggressive negotiations.
	option = {
		name = flavor_gbr.113.a
		if = {
			limit = {
				has_country_flag = eng_monarchy_in_absolute_power
			}
			event_target:revolt_province = {
				spawn_rebels = {
					type = parliamentarians_rebels
					size = 1
				}
			}
		}
		else_if = {
			limit = {
				has_country_flag = eng_parliament_in_absolute_power
			}
			event_target:revolt_province = {
				spawn_rebels = {
					type = pretender_rebels
					size = 1
				}
			}
		}
		else = {
			add_stability = -3
		}
		#freeze_government_power = {
		#	mechanic_type = parliament_vs_monarchy_mechanic
		#	power_type = governmental_power
		#}
		remove_government_reform = parliament_vs_monarchy_reform
		custom_tooltip = flavor_gbr.113.a.tt
		set_country_flag = eng_easier_civil_war_triggers
		ai_chance = {
			factor = 1
		}
	}
}

	}

		picture = PANIC_eventPicture
		trigger = {
			OR = {
				NOT = { has_reform = fifth_monarchists_reform }
				has_country_flag = fifth_monarchists_have_failed
			}
		}
	}
	desc = {
		desc = flavor_gbr.120.desc.b
		trigger = {
			has_reform = fifth_monarchists_reform
			NOT = { has_country_flag = fifth_monarchists_have_failed }
		}
	}
	picture = {
		picture = COMET_SIGHTED_eventPicture
		trigger = {
			has_reform = fifth_monarchists_reform
			NOT = { has_country_flag = fifth_monarchists_have_failed }
		}
	}
	is_triggered_only = yes
	trigger = {
		NOT = { has_country_flag = had_fifth_monarchists_end_event }
	}
	immediate = {
		hidden_effect = {
			set_country_flag = had_fifth_monarchists_end_event
		}
	}
	
	#The wheel has turned.
	option = {
		name = flavor_gbr.120.a
		if = {
			limit = {
				has_reform = fifth_monarchists_reform
				NOT = { has_country_flag = fifth_monarchists_have_failed }
			}
			set_government_power = {
				mechanic_type = end_time_mechanic
				power_type = fanatic_zeal
				value = 100
			}
			freeze_government_power = {
				mechanic_type = end_time_mechanic
				power_type = fanatic_zeal
			}
		}
		else = {
			kill_ruler = yes
			if = {
				limit = { has_heir = yes }
				kill_heir = {
					allow_new_heir = no
				}
			}
			change_government = monarchy
			add_government_reform = feudalism_reform
			release_all_possible_countries = yes
			set_bankruptcy = ROOT
			release_all_subjects = yes
		}
		ai_chance = {
			factor = 1
		}
	}
}
