	id = flavor_spa.20
	title = flavor_spa.20.t
	desc = flavor_spa.20.d
	picture = GOOD_WITH_MONARCH_eventPicture
	fire_only_once = yes 
	is_triggered_only = yes 
	immediate = {
		on_non_estate_disaster_end = yes
	}
	
	option = { 
		name = flavor_spa.20.a 
		
		add_legitimacy = 20 
		if = { 
			limit = { has_country_flag = cas_royalist_flag }
			change_estate_land_share = { 
				estate = all 
				share = -5
			}
		}
		else = { 
			add_estate_loyalty = {
				estate = estate_nobles 
				loyalty = 20 
			}
			custom_tooltip = spa_no_isabella_event_tt
			set_country_flag = no_isabella_event_flag
		}
		add_yearly_manpower = 2
		set_country_flag = ended_infantes_disaster_flag
	}    
}
