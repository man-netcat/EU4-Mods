# The fall of Silla (CK3: Hubaekje 892, Taebong 901, Goryeo 918, the surrender of 935).
# fall_of_silla.2 and .3 release the dormant Hubaekje and Taebong tags from the provinces
# Silla holds in their lands (the Dutch revolt pattern: cores, `release`, independence war).
namespace = fall_of_silla

country_event = {
	id = fall_of_silla.1
	title = fall_of_silla.1.t
	desc = fall_of_silla.1.d
	picture = CIVIL_WAR_eventPicture

	major = yes
	is_triggered_only = yes

	option = {
		name = fall_of_silla.1.a
		set_country_flag = had_fall_of_silla
	}
}

# Hubaekje: the south-west under Gyeon Hwon
country_event = {
	id = fall_of_silla.2
	title = fall_of_silla.2.t
	desc = fall_of_silla.2.d
	picture = CIVIL_WAR_eventPicture

	major = yes
	is_triggered_only = yes
	fire_only_once = yes

	trigger = {
		NOT = { exists = HUB }
		is_free_or_tributary_trigger = yes
		is_year = 892
		OR = {
			owns = 737
			owns = 4228
			owns = 4229
			owns = 1013
		}
	}

	immediate = {
		hidden_effect = {
			if = { limit = { 737 = { owned_by = ROOT } } 737 = { add_core = HUB } }
			if = { limit = { 4228 = { owned_by = ROOT } } 4228 = { add_core = HUB } }
			if = { limit = { 4229 = { owned_by = ROOT } } 4229 = { add_core = HUB } }
			if = { limit = { 1013 = { owned_by = ROOT } } 1013 = { add_core = HUB } }
			release = HUB
		}
	}

	option = {
		name = fall_of_silla.2.a
		HUB = {
			declare_war_with_cb = {
				who = ROOT
				casus_belli = cb_independence_war
			}
		}
	}
}
