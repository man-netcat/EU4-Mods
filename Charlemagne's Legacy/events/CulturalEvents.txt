# Support Commedia Dellarte
country_event = {
	id = cultural_event.1
	title = cultural_event.1.t
	desc = cultural_event.1.d
	picture = THEATRE_eventPicture

	trigger = {
		capital_scope = {
			continent = europe
		}
		religion_group = christian
		current_age = age_of_reformation
		NOT = {
			has_country_flag = commedia_dellarte
		}
		years_of_income = 0.1
		adm_power = 10
	}
	
	mean_time_to_happen = {
		months = 180
	}
	
	immediate = {
		set_country_flag = commedia_dellarte
	}
		
	option = {
		name = cultural_event.1.a
		add_country_modifier = {
			name = "commedia_dellarte"
			duration = 1460
		}
		add_adm_power = -10
		add_years_of_income = -0.1
	}
	option = {
		name = cultural_event.1.b
		add_prestige = -5
	}
}

	id = cultural_event.10
	title = cultural_event.10.t
	desc = cultural_event.10.d
	picture = ADVISOR_eventPicture

	trigger = {
		is_institution_enabled = renaissance
		capital_scope = { continent = europe }
		current_age = age_of_discovery
		NOT = {
			num_of_cities = 6
			has_country_flag = blooming_renaissance
		}
		advisor = philosopher
		adm_power = 30
		years_of_income = 0.3
		adm = 3
	}
	
	mean_time_to_happen = {
		months = 180
	}
	
	immediate = {
		set_country_flag = blooming_renaissance
	}
	
	option = {
		name = cultural_event.4.a
		capital_scope = {
			add_institution_embracement = {
				which = "renaissance"
				value = 100
			}
		}
		add_country_modifier = {
			name = blooming_renaissance
			duration = 7300
		}
		add_years_of_income = -0.3
		add_adm_power = -30
	}

		add_prestige = -10
	}
