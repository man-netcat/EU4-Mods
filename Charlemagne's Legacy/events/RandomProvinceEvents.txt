province_event = {
	id = 6032
	title = "EVTNAME6032"
	desc = "EVTDESC6032"
	picture = COURT_eventPicture
	goto = claim_province

	trigger = {
		owner = { religion = ROOT }
		owner = { is_subject  = no }
		any_neighbor_province = {
			NOT = { owned_by = ROOT }
			NOT = { is_core = ROOT }
			NOT = { is_claim = ROOT }
			religion = ROOT
			owner = {
				religion_group = ROOT
				NOT = { religion = ROOT }
				OR = {
					NOT = { adm = 2 }
					NOT = { dip = 2 }
					NOT = { mil = 2	}
				}
			}
			NOT = { owner = { is_subject_of = ROOT } }

		}
	}

	mean_time_to_happen = {
		months = 600
		
		modifier = {
			factor = 10
			owner = { historical_friend_with = ROOT }
		}
		modifier = {
			factor = 0.9
			owner = { historical_rival_with = ROOT }
		}		
		modifier = {
			factor = 1.2
			owner = { NOT = { dip = 1 } }
		}
		modifier = {
			factor = 2.0
			owner = { luck = yes }
		}
		modifier = {
			factor = 0.8
			owner = { is_defender_of_faith = yes }
		}
		modifier = {
			factor = 0.8
			owner = {
				dip = 5
			}
		}
	}

	is_mtth_scaled_to_size = 1

	immediate = {
		hidden_effect = {
			random_neighbor_province = {
				limit = {
					NOT = { owned_by = ROOT }
					NOT = { is_core = ROOT }
					NOT = { is_claim = ROOT }
					religion = ROOT
					owner = {
						religion_group = ROOT
						NOT = { religion = ROOT }
						OR = {
							NOT = { adm = 2 }
							NOT = { dip = 2 }
							NOT = { mil = 2	}
						}
					}
					NOT = { owner = { is_subject_of = ROOT } }
				}
				save_event_target_as = claim_province
			}
		}
	}

	option = {
		name = "EVTOPTA6032" # Press issue
		ai_chance = { factor = 55 }
		event_target:claim_province = {
			add_claim = ROOT
			owner = {

					target = ROOT
					type = cb_insult
					months = 12
				}
				add_opinion = { who = ROOT modifier = opinion_protect_faith }
			}
		}
	}
	option = {
		name = "EVTOPTB6032" # Settle issue
		ai_chance = { factor = 0 }
		owner = { add_stability = -1 }
	}
}
