# The fall of the Kirghiz Khaganate (CK3: the Khitan conquest, the title vacated on 925.1.2). Released by
# the disaster: the Naiman west of Mongolia, the Tuvans and the Sibir steppe become independent
# (cores, `release`, independence war - the Dutch revolt pattern); the Dzungarian counties pass to
# the existing Dzungaria.
namespace = fall_of_kirghiz

country_event = {
	id = fall_of_kirghiz.1
	title = fall_of_kirghiz.1.t
	desc = fall_of_kirghiz.1.d
	picture = CIVIL_WAR_eventPicture

	major = yes
	is_triggered_only = yes

	option = {
		name = fall_of_kirghiz.1.a
		set_country_flag = had_fall_of_kirghiz
	}
}

country_event = {
	id = fall_of_kirghiz.2
	title = fall_of_kirghiz.2.t
	desc = fall_of_kirghiz.2.d
	picture = CIVIL_WAR_eventPicture

	major = yes
	is_triggered_only = yes
	fire_only_once = yes

	trigger = {
		NOT = { exists = NIN }
		is_free_or_tributary_trigger = yes
		OR = {
			owns = 718
			owns = 4681
			owns = 4222
			owns = 2117
			owns = 4680
		}
	}

	immediate = {
		hidden_effect = {
			if = { limit = { 718 = { owned_by = ROOT } } 718 = { add_core = NIN } }
			if = { limit = { 4681 = { owned_by = ROOT } } 4681 = { add_core = NIN } }
			if = { limit = { 4222 = { owned_by = ROOT } } 4222 = { add_core = NIN } }
			if = { limit = { 2117 = { owned_by = ROOT } } 2117 = { add_core = NIN } }
			if = { limit = { 4680 = { owned_by = ROOT } } 4680 = { add_core = NIN } }
			release = NIN
		}
	}

	option = {
		name = fall_of_kirghiz.2.a
		NIN = {
			declare_war_with_cb = {
				who = ROOT
				casus_belli = cb_independence_war
			}
		}
	}
}
