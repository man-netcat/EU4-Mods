# The fall of the Kirghiz Khaganate (CK3: the Khitan conquest, the title vacated on 925.1.2). Released by
# the disaster: the Naiman west of Mongolia, the Tuvans and the Sibir steppe become independent
# (cores, `release`, independence war - the Dutch revolt pattern); the Dzungarian counties pass to
# the existing Dzungaria.
namespace = fall_of_kirghiz

country_event = {
	id = fall_of_kirghiz.2
	title = fall_of_kirghiz.2.t
	desc = fall_of_kirghiz.2.d
	picture = CIVIL_WAR_eventPicture

	major = yes
	is_triggered_only = yes
	fire_only_once = yes

	trigger = {
		NOT = { exists = NIN }
		is_free_or_tributary_trigger = yes
		OR = {
			owns = 718
			owns = 4681
			owns = 4222
			owns = 2117
			owns = 4680
		}
	}

	immediate = {
		hidden_effect = {
			718 = { limit = { owned_by = ROOT } add_core = NIN }
			4681 = { limit = { owned_by = ROOT } add_core = NIN }
			4222 = { limit = { owned_by = ROOT } add_core = NIN }
			2117 = { limit = { owned_by = ROOT } add_core = NIN }
			4680 = { limit = { owned_by = ROOT } add_core = NIN }
			release = NIN
		}
	}

	option = {
		name = fall_of_kirghiz.2.a
		NIN = {
			declare_war_with_cb = {
				who = ROOT
				casus_belli = cb_independence_war
			}
		}
	}
}
