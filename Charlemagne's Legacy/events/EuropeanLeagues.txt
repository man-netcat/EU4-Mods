###############################
# Catholic League
###############################
country_event = {
	id = league_events.6
	title = league_events.6.t
	desc = league_events.6.d
	picture = DIPLOMACY_eventPicture

	trigger = {
		NOT = { has_dlc = "Art of War" }
		religion = catholic
		is_part_of_hre = yes			
		is_religion_enabled = protestant
		is_year = 1605
		is_at_war = no
		NOT = { has_country_flag = catholic_league }
		NOT = { is_year = 1630 }
	}
	
	mean_time_to_happen = {
		months = 64
	}
	
	immediate = {
		set_country_flag = catholic_league 
	}
	
	option = {
		name = league_events.6.a
		ai_chance = { factor = 10 }
		every_country = {
			limit = {
				NOT = { tag = ROOT }
				religion = catholic
				is_part_of_hre = yes			
			}
			add_opinion = { who = ROOT modifier = catholic_league_catholic }
		}
		every_country = {
			limit = {
				OR = {
					religion = protestant
					religion = reformed
				}
			}
			add_opinion = { who = ROOT modifier = catholic_league_protestant }
		}		
		
	}
	option = {
		name = league_events.6.b
		ai_chance = { factor = 1 }
		add_prestige = -30

	}
}
