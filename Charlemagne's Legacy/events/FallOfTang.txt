# The fall of Tang (CK3: the Tang line ends 907, Dai Viet is independent from 938, the Sixteen
# Prefectures go to the Khitan in 936). fall_of_tang.2 releases the dormant Dai Viet tag (the
# Dutch revolt pattern: cores, `release`, independence war); .3 and .4 cede counties to tags
# that already exist, as the fall of the Kirghiz does for Dzungaria.
namespace = fall_of_tang

country_event = {
	id = fall_of_tang.1
	title = fall_of_tang.1.t
	desc = fall_of_tang.1.d
	picture = CIVIL_WAR_eventPicture

	major = yes
	is_triggered_only = yes

	option = {
		name = fall_of_tang.1.a
		set_country_flag = had_fall_of_tang
	}
}

# Dai Viet: the Jinghai circuit, whose Khuc governors rule it as their own
country_event = {
	id = fall_of_tang.2
	title = fall_of_tang.2.t
	desc = fall_of_tang.2.d
	picture = CIVIL_WAR_eventPicture

	major = yes
	is_triggered_only = yes
	fire_only_once = yes

	trigger = {
		NOT = { exists = DAI }
		is_free_or_tributary_trigger = yes
		is_year = 907
		OR = {
			owns = 613
			owns = 1016
			owns = 2371
			owns = 2372
			owns = 4819
			owns = 4820
			owns = 4821
		}
	}

	immediate = {
		hidden_effect = {
			if = { limit = { 613 = { owned_by = ROOT } } 613 = { add_core = DAI } }
			if = { limit = { 1016 = { owned_by = ROOT } } 1016 = { add_core = DAI } }
			if = { limit = { 2371 = { owned_by = ROOT } } 2371 = { add_core = DAI } }
			if = { limit = { 2372 = { owned_by = ROOT } } 2372 = { add_core = DAI } }
			if = { limit = { 4819 = { owned_by = ROOT } } 4819 = { add_core = DAI } }
			if = { limit = { 4820 = { owned_by = ROOT } } 4820 = { add_core = DAI } }
			if = { limit = { 4821 = { owned_by = ROOT } } 4821 = { add_core = DAI } }
			release = DAI
		}
	}

	option = {
		name = fall_of_tang.2.a
		DAI = {
			declare_war_with_cb = {
				who = ROOT
				casus_belli = cb_independence_war
			}
		}
	}
}

# The Sixteen Prefectures: the Khitan take the northern passes. The first Khitan tag in play
# (the great empire, the kingdom, then the duchy) receives them.
country_event = {
	id = fall_of_tang.4
	title = fall_of_tang.4.t
	desc = fall_of_tang.4.d
	picture = CIVIL_WAR_eventPicture

	major = yes
	is_triggered_only = yes
	fire_only_once = yes

	trigger = {
		is_free_or_tributary_trigger = yes
		is_year = 936
		OR = {
			exists = LIO
			exists = KIN
			exists = KTN
		}
		OR = {
			owns = 696
			owns = 697
			owns = 702
			owns = 1816
			owns = 2136
			owns = 4194
		}
	}

	immediate = {
		hidden_effect = {
			if = {
				limit = { exists = LIO }
				if = { limit = { 696 = { owned_by = ROOT } } 696 = { add_core = LIO cede_province = LIO } }
				if = { limit = { 697 = { owned_by = ROOT } } 697 = { add_core = LIO cede_province = LIO } }
				if = { limit = { 702 = { owned_by = ROOT } } 702 = { add_core = LIO cede_province = LIO } }
				if = { limit = { 1816 = { owned_by = ROOT } } 1816 = { add_core = LIO cede_province = LIO } }
				if = { limit = { 2136 = { owned_by = ROOT } } 2136 = { add_core = LIO cede_province = LIO } }
				if = { limit = { 4194 = { owned_by = ROOT } } 4194 = { add_core = LIO cede_province = LIO } }
			}
			else_if = {
				limit = { exists = KIN }
				if = { limit = { 696 = { owned_by = ROOT } } 696 = { add_core = KIN cede_province = KIN } }
				if = { limit = { 697 = { owned_by = ROOT } } 697 = { add_core = KIN cede_province = KIN } }
				if = { limit = { 702 = { owned_by = ROOT } } 702 = { add_core = KIN cede_province = KIN } }
				if = { limit = { 1816 = { owned_by = ROOT } } 1816 = { add_core = KIN cede_province = KIN } }
				if = { limit = { 2136 = { owned_by = ROOT } } 2136 = { add_core = KIN cede_province = KIN } }
				if = { limit = { 4194 = { owned_by = ROOT } } 4194 = { add_core = KIN cede_province = KIN } }
			}
			else = {
				if = { limit = { 696 = { owned_by = ROOT } } 696 = { add_core = KTN cede_province = KTN } }
				if = { limit = { 697 = { owned_by = ROOT } } 697 = { add_core = KTN cede_province = KTN } }
				if = { limit = { 702 = { owned_by = ROOT } } 702 = { add_core = KTN cede_province = KTN } }
				if = { limit = { 1816 = { owned_by = ROOT } } 1816 = { add_core = KTN cede_province = KTN } }
				if = { limit = { 2136 = { owned_by = ROOT } } 2136 = { add_core = KTN cede_province = KTN } }
				if = { limit = { 4194 = { owned_by = ROOT } } 4194 = { add_core = KTN cede_province = KTN } }
			}
		}
	}

	option = {
		name = fall_of_tang.4.a
	}
}
