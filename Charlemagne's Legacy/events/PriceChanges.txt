# Abolitionism
country_event = {
	id = prices.9
	title = prices.9.t
	desc = prices.9.d
	picture = TRADE_GOODS_PLANTATION_GOODS_IN_WAREHOUSE_eventPicture

	fire_only_once = yes
	
	major = yes

	trigger = {
		has_country_modifier = the_abolish_slavery_act
		NOT = { has_global_flag = abolitionism_happened }
	}
	mean_time_to_happen = {
		months = 1
	}
	
	
	option = {
		name = prices.9.a
		change_price = {
			trade_goods = slaves
			key = ABOLITIONISM
			value = -0.4
			duration = -1
		}
		add_prestige = 10
	}
}
