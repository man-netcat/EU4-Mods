# Cordoba cracks: the Emirate of the 860s holds Iberia by garrison and tribute, not consent.
# Each event fires once (country flag) after its own delay, so the pressure arrives over decades
# rather than at a stroke. Nothing here helps the Christians directly (see missions/Reconquista.txt);
# it only takes the emirate's grip away.
namespace = iberian_instability

# Muladi discontent: the convert families of the Marches and Extremadura resent Arab tax-farmers.
country_event = {
	id = kar_iberia.1
	title = kar_iberia_1_title
	desc = kar_iberia_1_desc
	picture = ANGRY_MOB_eventPicture

	trigger = {
		tag = ADU
		is_free_or_tributary_trigger = yes
		NOT = { has_country_flag = kar_iberia_muladi }
	}

	mean_time_to_happen = {
		months = 36
	}

	option = {
		name = kar_iberia_1_a
		ai_chance = { factor = 100 }
		set_country_flag = kar_iberia_muladi
		218 = { add_unrest = 4 }
		219 = { add_unrest = 4 }
		214 = { add_unrest = 4 }
		add_country_modifier = {
			name = kar_emirate_fraying
			duration = 3650
		}
	}

	option = {
		name = kar_iberia_1_b
		ai_chance = { factor = 0 }
		set_country_flag = kar_iberia_muladi
		add_years_of_income = -0.5
		218 = { add_unrest = 2 }
		219 = { add_unrest = 2 }
		214 = { add_unrest = 2 }
	}
}
