# Onin War
country_event = {
	id = flavor_jap.1
	title = "flavor_jap.EVTNAME1"
	desc = "flavor_jap.EVTDESC1"
	picture = BATTLE_eventPicture

	fire_only_once = yes
	
	trigger = {
		has_reform = daimyo
		any_country = {
			offensive_war_with = ROOT
			has_reform = daimyo
			is_in_war = {
				defender_leader = ROOT
				attacker_leader = THIS
			}
		}
	}

	immediate = {
		hidden_effect = {
			save_event_target_as = onin_defensive_war_leader
			random_country = {
				limit = {
					offensive_war_with = ROOT
					has_reform = daimyo
					is_in_war = {
						defender_leader = ROOT
						attacker_leader = THIS
					}
				}
				save_event_target_as = onin_offensive_war_leader
			}
			every_country = {
				limit = {
					NOT = { tag = ROOT }
					OR = {
						has_reform = daimyo
						has_reform = shogunate
					}
				}
				country_event = { id = flavor_jap.56 }
			}
		}
	}
	
	mean_time_to_happen = {
		days = 1
	}
	
	option = {
		name = "flavor_jap.EVTOPTA1"
		custom_tooltip = shogunate_mechanic_explanation
		add_country_modifier = {
			name = jap_sengoku_jidai
			duration = 10950
		}
	}
}

country_event = {
	id = flavor_jap.5
	title = "flavor_jap.EVTNAME5"
	desc = "flavor_jap.EVTDESC5"
	picture = SHIP_SAILING_eventPicture
	
	trigger = {
		has_country_modifier = sakoku_law
		is_subject = no
		had_country_flag = { 
			flag = sakoku_law
			days = 7300
		}
		NOT = { has_country_flag = jap_sakoku }
		NOT = {
			has_country_modifier = jap_follow_sakoku
		}
		NOT = {
			has_country_modifier = jap_question_sakoku
		}
	}
	
	mean_time_to_happen = {
		months = 600
	}
	
	option = {
		name = "flavor_jap.EVTOPTA5" # Keep following the Sakoku
		ai_chance = {
			factor = 100
		}
		add_country_modifier = {
			name = jap_follow_sakoku
			duration = 3650
		}
		set_country_flag = jap_sakoku
	}
	option = {
		name = "flavor_jap.EVTOPTB5" # Reject Sakoku law
		ai_chance = {
			factor = 0
		}
		add_country_modifier = {
			name = jap_question_sakoku
			duration = 3650
		}
		add_stability = -1
		remove_country_modifier = sakoku_law
		set_country_flag = jap_sakoku
		every_subject_country = {
			limit = {
				has_country_modifier = sakoku_law
			}
			remove_country_modifier = sakoku_law
		}
	}
}
