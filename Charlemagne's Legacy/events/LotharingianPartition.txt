# The partition of Lotharingia (history: Lothair II died at Piacenza on 8.8.869 with only an illegitimate son;
# Charles the Bald had himself crowned at Metz on 9.9.869; the Treaty of Meerssen of 8.8.870 divided the Middle Kingdom
# between him and Louis the German). CK3's 867 data has no partition (k_lotharingia goes whole to Charles on 869.8.8),
# so the line below is a deliberate, user-approved departure from CK3: Meuse and Rhone-Saone (Cambrai, Verdun, Toul,
# Besancon, Lyon, Vienne) to West Francia, the Rhineland, Frisia, Aachen, Metz and Alsace to East Francia.
# The tags LOT, HOL, KOL, TRI and DAU are the only ones touched; tools/checks/diplomacy.py keeps the lists below equal to
# the provinces those five tags own at the start.
# Flow: .1 (LOT, 869.8.9) -> accept: .4 after a year (the treaty, AI default); defy: .2 FRA and .3 GER claim their halves.
namespace = kar_lotharingia

# The death of Lothair II
country_event = {
	id = kar_lotharingia.1
	title = kar_lotharingia.1.t
	desc = kar_lotharingia.1.d
	picture = CIVIL_WAR_eventPicture

	major = yes
	fire_only_once = yes

	trigger = {
		tag = LOT
		date = 869.8.9
		OR = {
			owns = 1878
			owns = 188
			owns = 75
		}
	}

	mean_time_to_happen = {
		days = 3
	}

	# Defy the uncles: Hugh is proclaimed, the kingdom stays whole, and Charles and Louis both reach for it.
	option = {
		name = kar_lotharingia.1.a
		ai_chance = { factor = 0 }
		set_country_flag = kar_lotharingia_defiant
		add_country_modifier = {
			name = kar_lotharingia_succession_crisis
			duration = 1825
		}
		if = {
			limit = { exists = FRA }
			FRA = { country_event = { id = kar_lotharingia.2 days = 10 } }
		}
		if = {
			limit = { exists = GER }
			GER = { country_event = { id = kar_lotharingia.3 days = 10 } }
		}
	}

	# Treat with the uncles: the kingdom is divided at Meerssen a year on.
	option = {
		name = kar_lotharingia.1.b
		ai_chance = { factor = 100 }
		set_country_flag = kar_lotharingia_partitioning
		country_event = { id = kar_lotharingia.4 months = 12 }
	}
}

# The Treaty of Meerssen (8.8.870)

				92 = { limit = { owned_by = ROOT } remove_core = ROOT cede_province = FRA add_core = FRA }
				93 = { limit = { owned_by = ROOT } remove_core = ROOT cede_province = FRA add_core = FRA }

					GER = { vassalize = KOL }

					limit = { exists = TRI TRI = { is_subject_of = ROOT } }

				1878 = { limit = { owned_by = ROOT } remove_core = ROOT cede_province = GER add_core = GER }
				2975 = { limit = { owned_by = ROOT } remove_core = ROOT cede_province = GER add_core = GER }
				99 = { limit = { owned_by = ROOT } remove_core = ROOT cede_province = GER add_core = GER }
				98 = { limit = { owned_by = ROOT } remove_core = ROOT cede_province = GER add_core = GER }

				4721 = { limit = { owned_by = ROOT } remove_core = ROOT cede_province = GER add_core = GER }

			if = {
				limit = { exists = FRA }
				FRA = { add_prestige = 10 country_event = { id = kar_lotharingia.5 } }

				limit = { exists = GER }
