# Anatomical Theatre
country_event = {
	id = innovativeness_events.3
	title = "innovativeness_events.3.t"
	desc = "innovativeness_events.3.d"
	picture = ROYAL_SOCIETY_eventPicture
	
	goto = doctor_province
	
	trigger = {
		has_dlc = "Rule Britannia"
		innovativeness = 1
		any_owned_province = {
			is_state = yes
			is_capital = no
			#continent = CAPITAL
			NOT = {
				has_province_modifier = unpopular_dissections
			}
		}
	}
	
	is_triggered_only = yes
	
	immediate = {
		hidden_effect = {
			random_owned_province = {
				limit = {
					is_state = yes
					is_capital = no
					#continent = CAPITAL
					NOT = {
						has_province_modifier = unpopular_dissections
					}
				}
				save_event_target_as = doctor_province
			}
		}
	}
	
	mean_time_to_happen = {
		days = 1
		modifier = {
			factor = 1.2
			religion_group = christian
		}
	}
	
	option = {
		name = "innovativeness_events.3.a" #
		ai_chance = {
			factor = 20
		}
		reduce_innovativeness_small_effect = yes
	}
	option = {
		goto = doctor_province
		name = "innovativeness_events.3.b"
		ai_chance = {
			factor = 40
		}
		event_target:doctor_province = {
			area = {
				limit = { owned_by = ROOT }
				add_province_modifier = {
					name = "unpopular_dissections"
					duration = 1825
				}
			}
		}
	}
	option = {
		goto = doctor_province
		name = "innovativeness_events.3.c"
		ai_chance = {
			factor = 40
		}
		add_innovativeness_small_effect = yes
		add_prestige = -10
		event_target:doctor_province = {
			area = {
				limit = { owned_by = ROOT }
				add_province_modifier = {
					name = "unpopular_dissections"
					duration = 3650
				}
			}
		}
	}
}

# Promising Inventor
country_event = {
	id = innovativeness_events.4
	title = "innovativeness_events.4.t"
	desc = "innovativeness_events.4.d"
	picture = SCIENTISTS_eventPicture
	
	trigger = {
		has_dlc = "Rule Britannia"
		any_neighbor_country = {
			tech_difference = 1
		}
	}
	
	is_triggered_only = yes
	
	immediate = {
		hidden_effect = {
			random_neighbor_country = {
				limit = {
					tech_difference = 1
				}
				save_event_target_as = inventor_country
			}
			set_saved_name = {
				key = inventor
				type = advisor
				scope = event_target:inventor_country
			}
		}
	}
	
	mean_time_to_happen = {
		days = 1
	}
	
	option = {
		name = "innovativeness_events.4.a" #
		ai_chance = {
			factor = 40
		}
		if = {
			limit = {
				NOT = { government = native }
			}
			add_innovativeness_small_effect = yes
		}
		else_if = {
			limit = {
				government = native
			}
			add_adm_power = 50
		}
		event_target:inventor_country = {
			add_opinion = {
				who = ROOT
				modifier = harbored_inventor
			}
			every_ally = {
				limit = { NOT = { tag = ROOT } }
				add_opinion = {
					who = ROOT
					modifier = harbored_inventor
				}
			}
		}
	}
	option = {
		name = "innovativeness_events.4.b"
		ai_chance = {
			factor = 60
		}
		add_prestige = 10
		event_target:inventor_country = {
			add_opinion = {
				who = ROOT
				modifier = turned_away_inventor
			}
		}
	}
	after = {
		clear_saved_name = inventor
	}
}

	desc = "innovativeness_events.5.d"
	picture = ROYAL_SOCIETY_eventPicture
	goto = poor_province
	trigger = {
		has_dlc = "Rule Britannia"
		innovativeness = 1
		any_owned_province = {
			NOT = { development = 12 }
			is_capital = no
			#continent = CAPITAL
		}
	}
	
	is_triggered_only = yes
	
	immediate = {
		hidden_effect = {
			random_owned_province = {
				limit = {
					NOT = { development = 12 }
					is_capital = no
					#continent = CAPITAL
				}
				save_event_target_as = poor_province
			}
		}
	}
	
	mean_time_to_happen = {
		days = 1
	}
	
	option = {
		name = "innovativeness_events.5.a" #
		ai_chance = {
			factor = 40
		}
		add_innovativeness_small_effect = yes
		if = {
			limit = { has_estate = estate_church }
			add_estate_influence_modifier = {
				estate = estate_church
				desc = EST_VAL_PUBLIC_SCHOOLS
				influence = 10
				duration = 3650
			}
		}
		event_target:poor_province = {
			add_province_modifier = {
				name = "public_schools"
				duration = 10950
			}
		}
	}
	option = {
		name = "innovativeness_events.5.b"
		ai_chance = {
			factor = 60
		}
		reduce_estate_church_loyalty_effect = yes
		add_prestige = -5
	}
}
