	id = tengri_events.41
	title = tengri_events.41.t
	desc = tengri_events.41.d
	picture = RELIGIOUS_MIGRANTS_eventPicture
	
	trigger = {
		has_dlc = "The Cossacks"
		has_province_modifier = tengri_christian_mission
	}
	
	is_mtth_scaled_to_size = 1
	
	mean_time_to_happen = {
		months = 360
		modifier = {
			factor = 0.01
			religion = secondary
		}
	}
	
	option = {
		name = tengri_events.41.a #Praise the Lord
		remove_province_modifier = tengri_christian_mission
		if = {
			limit = {
				NOT = { religion = secondary }
			}
			change_to_secondary_religion = yes
		}
	}
}
