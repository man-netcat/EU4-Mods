# The Introduction of new diseases and exploitation
province_event = {
	id = 3073
	title = "EVTNAME3073"
	desc = "EVTDESC3073"
	picture = NATIVES_eventPicture

	trigger = {
		owner = { 
			OR = {
				has_country_flag = NF_ruthless
				native_policy = native_policy_hostile
			}
		}
		is_colony = yes
		NOT = { colonysize = 200 }
		native_size = 5
	}
	
	mean_time_to_happen = {
		months	= 240

		modifier = {
			factor = 0.8 
			NOT = { colonysize = 150 }
		}
		modifier = {
			factor = 0.8
			any_neighbor_province = {
				is_colony = yes
				owned_by = ROOT
			}
		}
	}
	
	goto = ROOT

	option = {
		name = "EVTOPTA3073"			# Ignore their suffering
		ai_chance = { factor = 75 }
		change_native_size = -5
		owner = {
			random_owned_province = {
				limit = {
					is_colony = yes
					native_size = 2
				}
				change_native_size = -2
			}
			add_country_modifier = {
				name = "disregard_natives"
				duration = 730
			}
		}
	}
	option = {
		name = "EVTOPTB3073"			# Attempt to assuage their pain
		ai_chance = { factor = 25 }
		owner = {
			add_treasury = -20
			add_country_modifier = {
				name = "aid_natives"
				duration = 365
			}
		}
	}
}

# Attempt to convert the natives.
province_event = {
	id = 3077
	title = "EVTNAME3077"
	desc = {
		trigger = {
			owner = {
				has_reform = mission_to_civilize_reform
			}
		}
		desc = "EVTDESC3077.C"
	}
	desc = {
		trigger = {
			owner = {
				OR = {
					has_country_flag = NF_peacenik
					native_policy = native_policy_coexist
				}
			}
		}
		desc = "EVTDESC3077.A"
	}
	desc = {
		trigger = {
			owner = {
				OR = {
					has_country_flag = NF_trader
					native_policy = native_policy_trade
				}
			}
		}
		desc = "EVTDESC3077.B"
	}
	picture = NATIVES_eventPicture
	
	trigger = {
		owner = {
			OR = {
				has_country_flag = NF_peacenik
				has_country_flag = NF_trader
				native_policy = native_policy_coexist
				native_policy = native_policy_trade
				has_reform = mission_to_civilize_reform
			}
		}

		OR = {
			AND = {
				religion = owner
				native_size = 1
			}
			NOT = { religion = owner }
		}
		is_colony = yes
		NOT = { native_hostileness = 4 }
		NOT = { has_province_modifier = converted_natives_colony_timer }
		NOT = {
			owner = {
				any_owned_province = {
					expulsion_target = {
						province_id = root
					}
				}
			}
		}
	}
	
	mean_time_to_happen = {
		months	= 320

		modifier = {
			factor = 0.9
			NOT = { native_hostileness = 3 }
		}
		modifier = {
			factor = 0.8
			NOT = { native_hostileness = 2 }
		}
		modifier = {
			factor = 0.9
			owner = { colonial_governor = 1 }
		}
	}

	immediate = {
		hidden_effect = {
			add_province_modifier = {
				name = converted_natives_colony_timer
				duration = 11000
				hidden = yes
			}
		}
	}
	
	goto = ROOT

	option = {
		name = "EVTOPTA3077"			# Try to convert of all them
		ai_chance = { factor = 70 }
		change_native_hostileness = 2
		change_native_ferocity = 2
		
		if = {
			limit = {
				NOT = { religion = owner }
			}
			change_religion = owner
		}
		else = {
			add_colonysize = 100
		}
	}
	
	option = {
		name = "EVTOPTB3077"			# Let them be
		ai_chance = { factor = 30 }
		change_native_hostileness = -2
		change_native_ferocity = -2
	}
}

### Native Accomodation ###
province_event = {
	id = 4901
	title = "EVTNAME4901"
	desc = "EVTDESC4901"
	picture = NATIVES_eventPicture
	
	trigger = {
		is_colony = yes
		native_size = 3
		native_ferocity = 1
		native_hostileness = 1
		NOT = { has_province_modifier = colonial_accommodation_mod }
		OR = {
			units_in_province = 1
			owner = { 
				OR = {
					has_country_flag = NF_peacenik
					native_policy = native_policy_coexist
				}
			}
		}
	}
	
	mean_time_to_happen = {
		years = 20
		modifier = {	
			factor = 0.9
			units_in_province = 2
			NOT = { native_size = 41 }
		}
		modifier = {	
			factor = 0.9
			units_in_province = 4
			NOT = { native_size = 81 }
		}
		modifier = {	
			factor = 0.9
			units_in_province = 6
			NOT = { native_size = 121 }
		}
		modifier = {	
			factor = 0.9	
			units_in_province = 8	
		}
		modifier = {	
			factor = 0.5	
			owner = { 
				OR = {
					has_country_flag = NF_peacenik
					native_policy = native_policy_coexist
				} 
			}
		}
	}
	
	goto = ROOT
	
	option = { 
		name = "EVTOPTA4901"
		if = {
			limit = {
				NOT = { has_dlc = "The Cossacks" }
			}
			change_native_ferocity = -1
		}
		add_province_modifier = {
			name = colonial_accommodation_mod
			duration = 3650
		}
		if = {
			limit = {
				owner = { 
					NOT = { has_country_modifier = colonial_assimilation_of_natives_mod } 
				}
			}
			owner = {
				add_country_modifier = {
					name = colonial_assimilation_of_natives_mod
					duration = 1825
				}
			}
		}
	}
}
