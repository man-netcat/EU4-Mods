# The fall of Balhae (CK3: the throne vacated in 926; by 1066 Jurchen tribes hold the Balhae counties).
# fall_of_balhae.2-.7 release one dormant Jurchen tribe each from the provinces Balhae holds in its
# lands (the Dutch revolt pattern: cores, `release`, independence war). Fired by the disaster.
namespace = fall_of_balhae

country_event = {
	id = fall_of_balhae.1
	title = fall_of_balhae.1.t
	desc = fall_of_balhae.1.d
	picture = CIVIL_WAR_eventPicture

	major = yes
	is_triggered_only = yes

	option = {
		name = fall_of_balhae.1.a
		set_country_flag = had_fall_of_balhae
	}
}

country_event = {
	id = fall_of_balhae.2
	title = fall_of_balhae.2.t
	desc = fall_of_balhae.2.d
	picture = CIVIL_WAR_eventPicture

	major = yes
	is_triggered_only = yes
	fire_only_once = yes

	trigger = {
		NOT = { exists = WNN }
		is_free_or_tributary_trigger = yes
		OR = {
			owns = 2188
			owns = 4217
			owns = 4654
		}
	}

	immediate = {
		hidden_effect = {
			2188 = { limit = { owned_by = ROOT } add_core = WNN }
			4217 = { limit = { owned_by = ROOT } add_core = WNN }
			4654 = { limit = { owned_by = ROOT } add_core = WNN }
			release = WNN
		}
	}

	option = {
		name = fall_of_balhae.2.a
		WNN = {
			declare_war_with_cb = {
				who = ROOT
				casus_belli = cb_independence_war
			}
		}
	}
}
