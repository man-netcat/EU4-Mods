# The fall of the Kirghiz Khaganate: CK3 vacates e_kirghiz_khanate on 925.1.2 (the Khitan conquest of
# the steppe). The Naiman west, Tuva and Sibir become independent and the Dzungarian counties pass to
# Dzungaria (events/FallOfKirghiz.txt); the Kirghiz keep Khakassia and the Ob country.
fall_of_kirghiz = {
	ended_by_country_breaking_to_rebels = no

	potential = {
		tag = KGZ
		NOT = { has_country_flag = had_fall_of_kirghiz }
		is_free_or_tributary_trigger = yes
		normal_or_historical_nations = yes
	}

	can_start = {
		is_year = 925
		has_any_disaster = no
	}

	can_stop = {
		has_any_disaster = yes
	}

	progress = {
		modifier = {
			factor = 2
			tag = KGZ
		}
		modifier = {
			factor = 3
			NOT = { stability = 1 }
		}
		modifier = {
			factor = 3
			is_at_war = yes
		}
		modifier = {
			factor = 4
			OR = {
				exists = LIO
				exists = KIN
			}
		}
	}

	can_end = {
		OR = {
			is_year = 960
			AND = {
				exists = NIN
				exists = TUV
				exists = SIB
			}
		}
	}

	modifier = {
		global_unrest = 2
		stability_cost_modifier = 0.25
	}

	on_start = fall_of_kirghiz.1
	on_end = fall_of_kirghiz.6

	on_monthly = {
		random_events = {
			240 = 0
			12 = fall_of_kirghiz.2
			12 = fall_of_kirghiz.3
			12 = fall_of_kirghiz.4
			12 = fall_of_kirghiz.5
		}
	}
}
